QUIZ ENGINI ALOQORITMI


class QuizEngine {
private:
    std::vector<int> shuffledIndices;
    int currentPos;
    int currentScore;
    QuizSet activeSet;
    std::mt19937 rng;
    
public:
    void startQuiz(const QuizSet& set) {
        activeSet = set;
        currentScore = 0;
        currentPos = 0;
        
        // Random tartib yaratish
        shuffledIndices.resize(set.questions.size());
        std::iota(shuffledIndices.begin(), shuffledIndices.end(), 0);
        std::shuffle(shuffledIndices.begin(), shuffledIndices.end(), rng);
    }
    
    Question getCurrentQuestion() {
        if (currentPos < shuffledIndices.size()) {
            return activeSet.questions[shuffledIndices[currentPos]];
        }
        throw std::out_of_range("No more questions");
    }
    
    std::vector<Answer> getShuffledAnswers() {
        auto q = getCurrentQuestion();
        auto answers = q.answers;
        std::shuffle(answers.begin(), answers.end(), rng);
        return answers;
    }
    
    bool submitAnswer(const Answer& selected) {
        bool correct = selected.isCorrect;
        if (correct) currentScore++;
        currentPos++;
        return correct;
    }
    
    bool hasMore() const {
        return currentPos < shuffledIndices.size();
    }
    
    QuizResult getResult() const {
        return {
            activeSet.id,
            currentScore,
            (int)activeSet.questions.size(),
            (currentScore * 100.0 / activeSet.questions.size()),
            time(nullptr),
            {}
        };
    }
};


 TXT PARSER MUKAMMAL LOGIKASI

 class TXTParser {
public:
    struct ParseResult {
        bool success;
        std::string quizName;
        std::vector<Question> questions;
        std::vector<std::string> errors;
        int lineNumber;
    };
    
    static ParseResult parse(const std::string& filename) {
        ParseResult result;
        result.success = true;
        result.lineNumber = 0;
        
        std::ifstream file(filename);
        if (!file.is_open()) {
            result.success = false;
            result.errors.push_back("Faylni ochib bo'lmadi!");
            return result;
        }
        
        // Fayl nomidan quiz nomi olish
        size_t lastSlash = filename.find_last_of("/\\");
        size_t lastDot = filename.find_last_of(".");
        if (lastSlash != std::string::npos) {
            result.quizName = filename.substr(lastSlash + 1, lastDot - lastSlash - 1);
        } else {
            result.quizName = filename.substr(0, lastDot);
        }
        
        std::string line;
        std::string currentQuestion;
        std::vector<Answer> currentAnswers;
        bool inQuestion = false;
        int questionCount = 0;
        
        while (std::getline(file, line)) {
            result.lineNumber++;
            
            // Trim whitespace
            line.erase(0, line.find_first_not_of(" \t"));
            line.erase(line.find_last_not_of(" \t") + 1);
            
            if (line.empty()) continue;
            
            if (line == "++++") {
                if (inQuestion && !currentQuestion.empty() && !currentAnswers.empty()) {
                    // Save previous question
                    Question q;
                    q.id = questionCount++;
                    q.text = currentQuestion;
                    q.answers = currentAnswers;
                    q.orderIndex = q.id;
                    result.questions.push_back(q);
                    
                    currentQuestion.clear();
                    currentAnswers.clear();
                }
                inQuestion = !inQuestion;
                continue;
            }
            
            if (inQuestion && line == "====") {
                continue;
            }
            
            if (inQuestion) {
                if (currentQuestion.empty() && line != "====") {
                    currentQuestion = line;
                } else if (!currentQuestion.empty()) {
                    // This is an answer
                    Answer a;
                    if (line[0] == '#') {
                        a.isCorrect = true;
                        a.text = line.substr(1);
                    } else {
                        a.isCorrect = false;
                        a.text = line;
                    }
                    
                    // Check duplicate correct answers
                    if (a.isCorrect) {
                        for (const auto& existing : currentAnswers) {
                            if (existing.isCorrect) {
                                result.errors.push_back("FAQAT Bitta to'gri javob bo'lishi kerak! Line: " + std::to_string(result.lineNumber));
                                result.success = false;
                            }
                        }
                    }
                    
                    currentAnswers.push_back(a);
                }
            }
        }
        
        // Validate answers count
        for (const auto& q : result.questions) {
            if (q.answers.size() < 2) {
                result.errors.push_back("Savol '" + q.text + "' da kamida 2 ta javob bolishi kerak!");
                result.success = false;
            }
            if (q.answers.size() > 6) {
                result.errors.push_back("Savol '" + q.text + "' da kopi bilan 6 ta javob bolishi mumkin!");
                result.success = false;
            }
            
            bool hasCorrect = false;
            for (const auto& a : q.answers) {
                if (a.isCorrect) hasCorrect = true;
            }
            if (!hasCorrect) {
                result.errors.push_back("Savol '" + q.text + "' da to'gri javob belgilanmagan (#) !");
                result.success = false;
            }
        }
        
        if (result.questions.empty()) {
            result.success = false;
            result.errors.push_back("Hech qanday savol topilmadi! Formatni tekshiring.");
        }
        
        file.close();
        return result;
    }
};

DATABASE MUKAMMAL IMPLEMENTATSIYASI

class Database {
private:
    sqlite3* db;
    std::mutex dbMutex;
    
    bool executeWithParams(const std::string& sql, std::function<void(sqlite3_stmt*)> binder) {
        std::lock_guard<std::mutex> lock(dbMutex);
        sqlite3_stmt* stmt;
        
        if (sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt, nullptr) != SQLITE_OK) {
            return false;
        }
        
        binder(stmt);
        
        bool success = (sqlite3_step(stmt) == SQLITE_DONE);
        sqlite3_finalize(stmt);
        return success;
    }
    
public:
    Database(const std::string& path) {
        sqlite3_open(path.c_str(), &db);
        createTables();
        createIndexes();
    }
    
    ~Database() {
        sqlite3_close(db);
    }
    
    void createTables() {
        const char* sql = R"(
            CREATE TABLE IF NOT EXISTS quiz_sets (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT UNIQUE NOT NULL,
                created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
            );
            
            CREATE TABLE IF NOT EXISTS questions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                quiz_set_id INTEGER NOT NULL,
                text TEXT NOT NULL,
                order_index INTEGER DEFAULT 0,
                FOREIGN KEY(quiz_set_id) REFERENCES quiz_sets(id) ON DELETE CASCADE
            );
            
            CREATE TABLE IF NOT EXISTS answers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                question_id INTEGER NOT NULL,
                text TEXT NOT NULL,
                is_correct INTEGER DEFAULT 0,
                FOREIGN KEY(question_id) REFERENCES questions(id) ON DELETE CASCADE
            );
            
            CREATE TABLE IF NOT EXISTS results (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                quiz_set_id INTEGER NOT NULL,
                score INTEGER NOT NULL,
                total INTEGER NOT NULL,
                percentage REAL NOT NULL,
                completed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY(quiz_set_id) REFERENCES quiz_sets(id)
            );
        )";
        
        executeSQL(sql);
    }
    
    void createIndexes() {
        executeSQL("CREATE INDEX IF NOT EXISTS idx_questions_set ON questions(quiz_set_id);");
        executeSQL("CREATE INDEX IF NOT EXISTS idx_answers_question ON answers(question_id);");
        executeSQL("CREATE INDEX IF NOT EXISTS idx_results_set ON results(quiz_set_id);");
        executeSQL("CREATE INDEX IF NOT EXISTS idx_results_date ON results(completed_at DESC);");
    }
    
    bool addQuizSet(const std::string& name, const std::vector<Question>& questions) {
        std::lock_guard<std::mutex> lock(dbMutex);
        
        sqlite3_exec(db, "BEGIN TRANSACTION", nullptr, nullptr, nullptr);
        
        // Insert quiz set
        sqlite3_stmt* stmt;
        const char* insertSet = "INSERT INTO quiz_sets (name) VALUES (?);";
        sqlite3_prepare_v2(db, insertSet, -1, &stmt, nullptr);
        sqlite3_bind_text(stmt, 1, name.c_str(), -1, SQLITE_STATIC);
        
        if (sqlite3_step(stmt) != SQLITE_DONE) {
            sqlite3_exec(db, "ROLLBACK", nullptr, nullptr, nullptr);
            sqlite3_finalize(stmt);
            return false;
        }
        
        int setId = sqlite3_last_insert_rowid(db);
        sqlite3_finalize(stmt);
        
        // Insert questions and answers
        for (const auto& q : questions) {
            const char* insertQ = "INSERT INTO questions (quiz_set_id, text, order_index) VALUES (?, ?, ?);";
            sqlite3_prepare_v2(db, insertQ, -1, &stmt, nullptr);
            sqlite3_bind_int(stmt, 1, setId);
            sqlite3_bind_text(stmt, 2, q.text.c_str(), -1, SQLITE_STATIC);
            sqlite3_bind_int(stmt, 3, q.orderIndex);
            
            if (sqlite3_step(stmt) != SQLITE_DONE) {
                sqlite3_exec(db, "ROLLBACK", nullptr, nullptr, nullptr);
                sqlite3_finalize(stmt);
                return false;
            }
            
            int qId = sqlite3_last_insert_rowid(db);
            sqlite3_finalize(stmt);
            
            for (const auto& a : q.answers) {
                const char* insertA = "INSERT INTO answers (question_id, text, is_correct) VALUES (?, ?, ?);";
                sqlite3_prepare_v2(db, insertA, -1, &stmt, nullptr);
                sqlite3_bind_int(stmt, 1, qId);
                sqlite3_bind_text(stmt, 2, a.text.c_str(), -1, SQLITE_STATIC);
                sqlite3_bind_int(stmt, 3, a.isCorrect ? 1 : 0);
                
                if (sqlite3_step(stmt) != SQLITE_DONE) {
                    sqlite3_exec(db, "ROLLBACK", nullptr, nullptr, nullptr);
                    sqlite3_finalize(stmt);
                    return false;
                }
                
                sqlite3_finalize(stmt);
            }
        }
        
        sqlite3_exec(db, "COMMIT", nullptr, nullptr, nullptr);
        return true;
    }
    
    std::vector<QuizSet> getAllQuizSets() {
        std::lock_guard<std::mutex> lock(dbMutex);
        std::vector<QuizSet> sets;
        
        sqlite3_stmt* stmt;
        const char* sql = R"(
            SELECT qs.id, qs.name,
                   COUNT(DISTINCT q.id) as question_count
            FROM quiz_sets qs
            LEFT JOIN questions q ON qs.id = q.quiz_set_id
            GROUP BY qs.id
            ORDER BY qs.created_at DESC;
        )";
        
        if (sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr) == SQLITE_OK) {
            while (sqlite3_step(stmt) == SQLITE_ROW) {
                QuizSet set;
                set.id = sqlite3_column_int(stmt, 0);
                set.name = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
                sets.push_back(set);
            }
        }
        sqlite3_finalize(stmt);
        
        return sets;
    }
    
    QuizSet getFullQuizSet(int id) {
        QuizSet set;
        set.id = id;
        
        // Get name
        sqlite3_stmt* stmt;
        const char* getName = "SELECT name FROM quiz_sets WHERE id = ?;";
        sqlite3_prepare_v2(db, getName, -1, &stmt, nullptr);
        sqlite3_bind_int(stmt, 1, id);
        
        if (sqlite3_step(stmt) == SQLITE_ROW) {
            set.name = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
        }
        sqlite3_finalize(stmt);
        
        // Get questions
        const char* getQuestions = "SELECT id, text, order_index FROM questions WHERE quiz_set_id = ? ORDER BY order_index;";
        sqlite3_prepare_v2(db, getQuestions, -1, &stmt, nullptr);
        sqlite3_bind_int(stmt, 1, id);
        
        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Question q;
            q.id = sqlite3_column_int(stmt, 0);
            q.text = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            q.orderIndex = sqlite3_column_int(stmt, 2);
            
            // Get answers for this question
            sqlite3_stmt* ansStmt;
            const char* getAnswers = "SELECT id, text, is_correct FROM answers WHERE question_id = ?;";
            sqlite3_prepare_v2(db, getAnswers, -1, &ansStmt, nullptr);
            sqlite3_bind_int(ansStmt, 1, q.id);
            
            while (sqlite3_step(ansStmt) == SQLITE_ROW) {
                Answer a;
                a.id = sqlite3_column_int(ansStmt, 0);
                a.text = reinterpret_cast<const char*>(sqlite3_column_text(ansStmt, 1));
                a.isCorrect = sqlite3_column_int(ansStmt, 2) == 1;
                q.answers.push_back(a);
            }
            sqlite3_finalize(ansStmt);
            
            set.questions.push_back(q);
        }
        sqlite3_finalize(stmt);
        
        return set;
    }
    
    bool deleteQuizSet(int id) {
        std::lock_guard<std::mutex> lock(dbMutex);
        std::string sql = "DELETE FROM quiz_sets WHERE id = " + std::to_string(id) + ";";
        return executeSQL(sql);
    }
    
    bool saveResult(const QuizResult& result) {
        std::lock_guard<std::mutex> lock(dbMutex);
        const char* sql = "INSERT INTO results (quiz_set_id, score, total, percentage) VALUES (?, ?, ?, ?);";
        
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);
        sqlite3_bind_int(stmt, 1, result.quizSetId);
        sqlite3_bind_int(stmt, 2, result.score);
        sqlite3_bind_int(stmt, 3, result.total);
        sqlite3_bind_double(stmt, 4, result.percentage);
        
        bool success = (sqlite3_step(stmt) == SQLITE_DONE);
        sqlite3_finalize(stmt);
        return success;
    }
    
    std::vector<QuizResult> getTopResults() {
        std::vector<QuizResult> results;
        const char* sql = "SELECT quiz_set_id, score, total, percentage, completed_at FROM results ORDER BY percentage DESC LIMIT 10;";
        
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);
        
        while (sqlite3_step(stmt) == SQLITE_ROW) {
            QuizResult r;
            r.quizSetId = sqlite3_column_int(stmt, 0);
            r.score = sqlite3_column_int(stmt, 1);
            r.total = sqlite3_column_int(stmt, 2);
            r.percentage = sqlite3_column_double(stmt, 3);
            r.completedAt = sqlite3_column_int(stmt, 4);
            results.push_back(r);
        }
        sqlite3_finalize(stmt);
        
        return results;
    }
    
private:
    bool executeSQL(const std::string& sql) {
        char* errMsg = nullptr;
        int rc = sqlite3_exec(db, sql.c_str(), nullptr, nullptr, &errMsg);
        
        if (rc != SQLITE_OK) {
            std::cerr << "SQL xato: " << errMsg << std::endl;
            sqlite3_free(errMsg);
            return false;
        }
        return true;
    }
};
